  justification=raggedright,singlelinecheck=false,position=top,skip=8pt}
\newcommand{\floatcaption}[2]{%
  \caption[#1]{\textbf{#1}%
    \if\relax\detokenize{#2}\relax\else.\space{\normalfont\footnotesize #2}\fi}%
}
